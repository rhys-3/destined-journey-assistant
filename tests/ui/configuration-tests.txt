  check('删除自定义模型只移除三个所属条目',!ui.state.config.custom_models.some(m=>m.id===gid)&&glm.ids.every(id=>!data.prompts.some(p=>p.id===id))&&alt.ids.every(id=>data.prompts.some(p=>p.id===id)));
  check('删除模型不修改已保存配置',ui.configLibrary().items.find(i=>i.id===idA).snapshot.preset.config.custom_models.some(m=>m.id===gid));
  await ui.applyConfiguration(idA);
  check('恢复旧配置能够恢复已删除模型',ui.state.config.custom_models.some(m=>m.id===gid)&&glm.ids.every(id=>data.prompts.some(p=>p.id===id)));
  const withoutSource=structuredClone(data);
  data.prompts=data.prompts.filter(p=>p.id!==builtin.Gemini.ids[0]);ui.reconcilePreset();
  const countBeforeMissing=data.prompts.length;
  ui.state.editorUnlocked=true;await rejected('Gemini 来源缺失停止创建',()=>ui.addCustomModel('不应新增'),'模板缺失');
  check('来源缺失没有半创建模型',data.prompts.length===countBeforeMissing&&!ui.state.config.custom_models.some(m=>m.label==='不应新增'));
  data=withoutSource;ui.reconcilePreset();

  // Keep a populated configuration view for layout checks; restore external APIs.
  for(const [key,value] of Object.entries(apis))window[key]=value;
  ui.state.config.connection_link.enabled=false;
  ui.state.config.configuration_library.pendingWorld=null;
  window.replaceVariables(ui.state.config);
  ui.state.activeTab='configurations';ui.render();
  check('配置页面和顶部快捷切换完整',!!ui.shadow.querySelector('[data-action="configuration-save-new"]')&&ui.shadow.querySelectorAll('[data-action="configuration-switch"] option').length===ui.configLibrary().items.length+1);
  return results;
